```typescript
// Add detectAngularFramework function near detectNextFramework (around line 389)
const detectAngularFramework = (cwd: string, filePath: string): 'angular' | undefined => {
  // 1. package.json @angular/core dep
  try {
    const pkg = JSON.parse(fs.readFileSync(path.resolve(cwd, 'package.json'), 'utf8'));
    const deps = { ...pkg.dependencies, ...pkg.devDependencies, ...pkg.peerDependencies };
    if (deps['@angular/core']) return 'angular';
  } catch {}
  // 2. angular.json at root
  if (fs.existsSync(path.resolve(cwd, 'angular.json'))) return 'angular';
  // 3. Standard Angular structure markers
  const markers = ['src/app/app.component.ts', 'src/main.ts'];
  for (const marker of markers) {
    if (fs.existsSync(path.resolve(cwd, marker))) return 'angular';
  }
  // 4. Bounded depth walk for .component.ts / .service.ts (depth 3)
  const srcAppDir = path.resolve(cwd, 'src/app');
  if (fs.existsSync(srcAppDir)) {
    try {
      const files = fs.readdirSync(srcAppDir, { recursive: true, maxDepth: 3 });
      const hasAngularFiles = files.some(f => 
        f.endsWith('.component.ts') || f.endsWith('.service.ts')
      );
      if (hasAngularFiles) return 'angular';
    } catch {}
  }
  return undefined;
};

// In buildEvidenceOutput, update framework detection chain:
const framework = detectAngularFramework(cwd, fn.file) 
  ?? detectNextFramework(cwd, fn.file) 
  ?? undefined;
```
